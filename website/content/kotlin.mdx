    kotlin("jvm") version "2.2.20"
}

dependencies {
    implementation("io.github.marinasundstrom.myservicebus:myservicebus-kotlin:0.1.0-preview.9")
    implementation("io.github.marinasundstrom.myservicebus:myservicebus-rabbitmq:0.1.0-preview.9")

}`;

export const kotlinConsumer = `data class SubmitOrder(val orderId: UUID)
data class OrderSubmitted(val orderId: UUID)

class SubmitOrderConsumer : Consumer<SubmitOrder> {
    override suspend fun consume(context: ConsumeContext<SubmitOrder>) {
        context.publish(OrderSubmitted(context.message.orderId))
    }
}`;

export const configureAndPublish = `val services = ServiceCollection.create()

services.addServiceBus {
    consumer<SubmitOrderConsumer>()
    transport<RabbitMqFactoryConfigurator> { context ->
        host("localhost")
        configureEndpoints(context)
    }
}

val provider = services.buildServiceProvider()
val bus = provider.getRequiredService<MessageBus>()
bus.start()

runBlocking {
    val order = SubmitOrder(UUID.randomUUID())
    bus.publish(order) {
        correlationId = order.orderId
        headers["tenant"] = "north"
    }
}`;

export const requestClient = `provider.createScope().use { scope ->
    val factory = scope.serviceProvider
        .getRequiredService<RequestClientFactory>()
    val client = factory.create<LookupOrder>(timeout = 10.seconds)

    val result: RequestResult<OrderStatus, OrderNotFound> =
        client.requestOneOf(LookupOrder(orderId))

    when (result) {
        is RequestResult.First -> println(result.message.status)
        is RequestResult.Second -> println("Missing \${result.message.orderId}")
    }
}`;

export const mediator = `class LookupOrderHandler : Handler<LookupOrder, OrderStatus> {
    override suspend fun handle(context: ConsumeContext<LookupOrder>): OrderStatus =
        findOrder(context.message.orderId)
}

val mediator = ServiceCollection.create().createMediator {
    handler<LookupOrderHandler>()
}

val status: OrderStatus = mediator.request(LookupOrder(orderId))`;

export const ktorIntegration = `fun Application.messagingModule() {
    install(MyServiceBus) {
        bus {
            consumer<LookupOrderConsumer>()
            transport<RabbitMqFactoryConfigurator> { context ->
                host("localhost")
                configureEndpoints(context)
            }
        }
        stopTimeout = 30.seconds
    }

    routing {
        post("/orders/{orderId}/publish") {
            val orderId = UUID.fromString(call.parameters["orderId"])
            call.myServiceBus.publish(SubmitOrder(orderId))
            call.respond(HttpStatusCode.Accepted)
        }

        get("/orders/{orderId}") {
            val orderId = UUID.fromString(call.parameters["orderId"])
            val status: OrderStatus =
                call.myServiceBus.request(LookupOrder(orderId), timeout = 10.seconds)
            call.respond(status)
        }
    }
}`;

# Kotlin is a first-class JVM target.

<p className="docs-summary">The Kotlin projection rebuilds MyServiceBus around coroutines, receiver DSLs, reified types, Kotlin durations, and exhaustive results. It shares the Java transport and delivery runtime without inheriting Java futures or overload-heavy API shapes.</p>

<div className="callout callout-accent">
  <strong>Experimental Kotlin API</strong>
  <p>The shared JVM runtime and Java projection are the stable reference surfaces for this preview. The Kotlin projection passes the same delivery and staged-package gates, but its source API may evolve between previews as Kotlin and framework conventions are refined.</p>
</div>

## Install the projection

The Kotlin artifact sits above the shared JVM runtime. Add the Kotlin Jackson module when messages use data-class primary constructors.

## Consume with suspending code

Kotlin owns both <code>Consumer&lt;T&gt;</code> and <code>ConsumeContext&lt;T&gt;</code>. The handler can call suspending messaging operations directly; MyServiceBus waits for the coroutine before acknowledging the delivery and propagates cancellation and failure into the shared receive pipeline.

The mediator uses the shared scoped consumer pipeline but stays entirely in process. A suspending handler returns its response directly.

## Host it natively in Ktor

The server sample integrates through a Ktor application plugin instead of a Java-shaped bootstrap wrapper. One <code>install(MyServiceBus)</code> block declares the bus and optional application services; the plugin builds and owns the provider, startup, graceful shutdown with Kotlin <code>Duration</code>, readiness, and scopes. Suspending routes use the projected bus or open a scoped service block directly. The generic adapter remains sample-first while its ownership model settles. Its integration test runs publish, directed send, and correlated request/response through an isolated RabbitMQ broker and waits for the consumers to complete.
