export const installKotlin = `plugins {
    kotlin("jvm") version "2.2.20"
}

dependencies {
    implementation("io.github.marinasundstrom.myservicebus:myservicebus-kotlin:VERSION")
    implementation("io.github.marinasundstrom.myservicebus:myservicebus-rabbitmq:VERSION")
    implementation("com.fasterxml.jackson.module:jackson-module-kotlin:2.17.2")
}`;

export const kotlinConsumer = `data class SubmitOrder(val orderId: UUID)
data class OrderSubmitted(val orderId: UUID)

class SubmitOrderConsumer : Consumer<SubmitOrder> {
    override suspend fun consume(context: ConsumeContext<SubmitOrder>) {
        context.publish(OrderSubmitted(context.message.orderId))
    }
}`;

export const configureAndPublish = `val services = ServiceCollection.create()

services.addServiceBus {
    consumer<SubmitOrderConsumer>()
    transport<RabbitMqFactoryConfigurator> { context ->
        host("localhost")
        configureEndpoints(context)
    }
}

val provider = services.buildServiceProvider()
val bus = provider.getRequiredService<MessageBus>()
bus.start()

runBlocking {
    val order = SubmitOrder(UUID.randomUUID())
    bus.publish(order) {
        correlationId = order.orderId
        headers["tenant"] = "north"
    }
}`;

export const requestClient = `provider.createScope().use { scope ->
    val factory = scope.serviceProvider
        .getRequiredService<RequestClientFactory>()
    val client = factory.create<LookupOrder>(timeout = 10.seconds)

    val result: RequestResult<OrderStatus, OrderNotFound> =
        client.requestOneOf(LookupOrder(orderId))

    when (result) {
        is RequestResult.First -> println(result.message.status)
        is RequestResult.Second -> println("Missing \${result.message.orderId}")
    }
}`;

export const mediator = `class LookupOrderHandler : SuspendHandler<LookupOrder, OrderStatus> {
    override suspend fun handle(request: LookupOrder): OrderStatus =
        findOrder(request.orderId)
}

val mediator = ServiceCollection.create().createMediator {
    handler<LookupOrderHandler>()
}

val status: OrderStatus = mediator.request(LookupOrder(orderId))`;

# Kotlin is a first-class JVM target.

<p className="docs-summary">The Kotlin projection rebuilds MyServiceBus around coroutines, receiver DSLs, reified types, Kotlin durations, and exhaustive results. It shares the Java transport and delivery runtime without inheriting Java futures or overload-heavy API shapes.</p>

<div className="callout callout-accent">
  <strong>Experimental until the MVP gate is complete</strong>
  <p>This projection is being developed on its own feature branch and is not yet an announced stable surface. Breaking changes are expected while broker flows, scope behavior, cancellation, the runnable sample, and staged-package verification converge.</p>
</div>

## Install the projection

The Kotlin artifact sits above the shared JVM runtime. Add the Kotlin Jackson module when messages use data-class primary constructors. Replace <code>VERSION</code> with the preview version that first includes the Kotlin MVP.

<CodeViewer code={installKotlin} label="Gradle Kotlin DSL" language="kotlin" />

## Consume with suspending code

Kotlin owns both <code>Consumer&lt;T&gt;</code> and <code>ConsumeContext&lt;T&gt;</code>. The handler can call suspending messaging operations directly; MyServiceBus waits for the coroutine before acknowledging the delivery and propagates cancellation and failure into the shared receive pipeline.

<CodeViewer code={kotlinConsumer} label="Suspending Kotlin consumer" language="kotlin" />

## Configure and publish without Java ceremony

The registration DSL infers consumer and transport classes. The projected bus publishes with the familiar name instead of an <code>Await</code> suffix, while configured operations receive Kotlin-owned metadata contexts.

## Request with reified response types

The scoped factory accepts Kotlin <code>Duration</code> and creates a suspending client without class literals. Multiple business outcomes become a sealed result, so <code>when</code> remains exhaustive and retains branch identity.
